   137	# Transition mutex (1.1.4, r5): ONE flock (fd 7 on /tmp/mk20/transition.lock) is held for the WHOLE setup or the WHOLE
   138	# rollback, so they can never overlap (flapmon, the confirm waiter and mk20d may all ask for --rollback-now). Every
   139	# process spawned while it is held gets 7>&- (adbd, bridge loop, KD, add-adb, flapmon, confirm waiter), so only the
   140	# owner holds it and it is released when the owner exits. A rollback that finds stock verified is a no-op: exit 0
   141	# always means stock is up. Every lock step is checked (fail closed: no mutation without the mutex).
   142	# Owner-local bounded supervisor: the rollback runs cleanup in its own child and waits at most MK20_RB_TIMEOUT
   143	# (default 240 s); on timeout it stops, kills and reaps ITS OWN child tree before releasing the mutex. No caller ever
   144	# kills another owner; a waiter that cannot get the mutex in MK20_RB_LOCK_TRIES*0.25 s (default 300 s) gives up (rc 1).

   147	stock_ok() { [ -e $C/acm.usb0 ] && [ ! -e $C/ecm.usb0 ] && [ ! -e $C/hid.usb0 ] && [ -n "$(udc_now)" ] &&

   152	# tl_lock <tries>: take the transition mutex on fd 7 (tries 0.25 s apart; BusyBox flock has no -w). 1 = not held.
   153	tl_lock() {
   154	  mkdir -p "$(dirname "$TL")" 2>/dev/null || return 1
   155	  touch "$TL" 2>/dev/null || return 1   # an exec redirect error would exit the shell (special builtin)
   156	  exec 7>"$TL" || return 1
   157	  n=0; until flock -n 7; do n=$((n+1)); [ "$n" -ge "$1" ] && { exec 7>&-; return 1; }; sleep 0.25; done
   158	}
   159	kids() { for d in /proc/[0-9]*; do [ "$(awk '/^PPid:/{print $2}' "$d/status" 2>/dev/null)" = "$1" ] && echo "${d#/proc/}"; done; }
   160	kill_tree() { kill -STOP "$1" 2>/dev/null; for k in $(kids "$1"); do kill_tree "$k"; done; kill -9 "$1" 2>/dev/null; }
   161	# uint_cap <value> <default> <min> <max>: a validated, clamped decimal (no octal, no overflow: >6 digits = max).
   162	# Every timeout/lock-try knob goes through this BEFORE any lock or fork (r5: bad arithmetic must not kill the owner).
   163	uint_cap() {
   164	  case "$1" in ''|*[!0-9]*) echo "$2"; return;; esac
   165	  v=${1#"${1%%[!0]*}"}; [ -n "$v" ] || v=0
   166	  [ ${#v} -gt 6 ] && { echo "$4"; return; }
   167	  [ "$v" -lt "$3" ] && v=$3; [ "$v" -gt "$4" ] && v=$4; echo "$v"
   168	}
   169	# r5 #5: cleanup worst case on a HEALTHY board = kd_stop<=32 + umount<=10 + kd_start<=23 + readd_adb (g1 wait<=60,
   170	# 5, add-adb ~4 x7, 6x15 re-add window) ~= 260 s -> 330 s so a healthy rollback incl. add-adb is never cut off.
   171	RBT=$(uint_cap "${MK20_RB_TIMEOUT:-}" 330 1 900)          # rollback / post-timeout cleanup bound (s)
   172	RBTRIES=$(uint_cap "${MK20_RB_LOCK_TRIES:-}" 3600 1 4800)  # rollback mutex wait (x0.25 s) >= SUT+RBT+30 s owner hold
   173	SUTRIES=$(uint_cap "${MK20_SETUP_LOCK_TRIES:-}" 40 1 4800) # setup mutex wait (x0.25 s)
   174	SUT=$(uint_cap "${MK20_SETUP_TIMEOUT:-}" 450 1 1800)       # whole setup incl. its own failsafe (~60 s + cleanup <=330 s)
   175	# bounded_wait <pid> <s>: 0 = the child exited in time (rc in BW_RC); 1 = timed out: its whole tree was stopped,
   176	# SIGKILLed and the child reaped; 2 = timed out AND still alive 10 s after SIGKILL (D state, e.g. a stuck UDC write):
   177	# not waited for (an unbounded wait would wedge us). Only ever used on OUR OWN children.
   178	bounded_wait() {
   179	  bst=$(pstart "$1"); bi=0; BW_RC=1
   180	  while [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ] && [ $bi -lt $(($2*4)) ]; do sleep 0.25; bi=$((bi+1)); done
   181	  if [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ]; then
   182	    kill_tree "$1"; bi=0
   183	    while [ "$(pstart "$1")" = "$bst" ] && [ $bi -lt 40 ]; do sleep 0.25; bi=$((bi+1)); done
   184	    [ "$(pstart "$1")" = "$bst" ] && return 2
   185	    wait "$1" 2>/dev/null; return 1
   186	  fi
   187	  wait "$1" 2>/dev/null; BW_RC=$?; return 0
   188	}

=== firmware/release/share/flapmon.sh
```sh
     1	#!/bin/sh
     2	# MK20 flap monitor (1.1.2+; was an inline heredoc in gadget.sh). Started detached by gadget.sh:
     3	#   flapmon.sh <udc-state-file> <window_s> <drops> <log> <stopfile> -- <rollback command...>
     4	# Counts link drops; >= <drops> within <window_s> -> touch the gadget.busy marker and run the rollback command once,
     5	# then exit. A drop is: configured -> anything but configured/suspended, or a SHORT suspend (< MK20_FLAP_SUSP_S,
     6	# default 5 s) that ends in anything but configured (a bus reset/glitch the poll caught as idle). A long suspend
     7	# (Mac sleep) is not a flap: suspend, resume and a resume reset / unplug after it are logged, not counted.
     8	# Single instance (flock on fd 8, not inherited by the rollback). The stopfile is re-checked right before rollback.
     9	# The busy marker is cleared after every rollback attempt. A failed rollback is retried (MK20_FLAP_TRIES, default 3,
    10	# MK20_FLAP_RETRY_S apart); if all fail, the persistent mk20-no-composite flag is set and the watcher hard-resets
    11	# (cmd-reboot), so the next boot runs stock until the flag is deleted.
    12	# The rollback command bounds itself (gadget.sh --rollback-now: transition mutex + owner-local supervisor that kills
    13	# and reaps its own cleanup tree after MK20_RB_TIMEOUT), so a hung rollback returns rc!=0 here and is retried/escalated.
    14	# Writes the ready marker (MK20_FLAP_READY, default /tmp/mk20/flapmon.ready) once the lock is held.
    15	# Test env: MK20_UPTIME_CMD (seconds), MK20_FLAP_POLL (default 0.3), MK20_BUSY, MK20_FLAP_LOCK, MK20_FLAP_SUSP_S.
    16	[ $# -ge 7 ] && [ "$6" = -- ] || { echo "usage: flapmon.sh state window drops log stopfile -- cmd..." >&2; exit 2; }
    17	U=$1; W=$2; N=$3; LOG=$4; STOP=$5; shift 6
    18	case "$W$N" in ''|*[!0-9]*) echo "flapmon: window/drops must be integers" >&2; exit 2;; esac
    19	BUSY=${MK20_BUSY:-/tmp/mk20/gadget.busy}; SUSP=${MK20_FLAP_SUSP_S:-5}
    20	LOCK=${MK20_FLAP_LOCK:-/tmp/mk20/flapmon.lock}; mkdir -p "$(dirname "$LOCK")"
    21	exec 8>"$LOCK"; i=0; while ! flock -n 8; do i=$((i+1)); [ $i -lt 10 ] && { sleep 0.5; continue; }
    22	  { echo "flapmon $(cut -d. -f1 /proc/uptime): another monitor holds $LOCK" >> "$LOG"; exit 3; }; done
    23	READY=${MK20_FLAP_READY:-/tmp/mk20/flapmon.ready}; mkdir -p "$(dirname "$READY")"; : > "$READY"   # lock held: gadget.sh waits for this
    24	up() { if [ -n "${MK20_UPTIME_CMD:-}" ]; then $MK20_UPTIME_CMD; else cut -d. -f1 /proc/uptime; fi; }
    25	prev=$(cat "$U" 2>/dev/null); times=""; sus=""
    26	while [ ! -f "$STOP" ]; do
    27	  s=$(cat "$U" 2>/dev/null); now=""
    28	  if [ "$s" != "$prev" ]; then
    29	    if [ "$prev" = configured ] && [ "$s" = suspended ]; then
    30	      sus=$(up); echo "flapmon $sus: host suspend" >> "$LOG"
    31	    elif [ "$prev" = configured ]; then
    32	      now=$(up)
    33	    elif [ "$prev" = suspended ]; then
    34	      t=$(up); d=$((t - ${sus:-$t}))
    35	      if [ "$s" = configured ]; then echo "flapmon $t: host resume after ${d}s" >> "$LOG"
    36	      elif [ -n "$sus" ] && [ "$d" -lt "$SUSP" ]; then now=$t
    37	      else echo "flapmon $t: '$s' after ${d}s suspend (resume reset/unplug, not counted)" >> "$LOG"; fi
    38	      sus=""
    39	    fi
    40	  fi
    41	  if [ -n "$now" ]; then
    42	    nt=""; cnt=0
    43	    for t in $times $now; do [ $((now-t)) -le "$W" ] && { nt="$nt $t"; cnt=$((cnt+1)); }; done; times=$nt
    44	    echo "flapmon $now: link drop ($s), $cnt in ${W}s" >> "$LOG"
    45	    if [ "$cnt" -ge "$N" ]; then
    46	      [ -f "$STOP" ] && exit 0
    47	      echo "flapmon $now: FLAPPING -> rollback" >> "$LOG"
    48	      try=1
    49	      while :; do
    50	        # r5 #6: only clear a marker we created, and never after a refusal (rc 4: another owner holds the mutex and
    51	        # owns gadget.busy; it clears it itself under the mutex). rc 5 = stale generation: our composite is gone.
    52	        mine=0; [ -e "$BUSY" ] || { touch "$BUSY"; mine=1; }
    53	        "$@" 8>&-; rc=$?
    54	        [ "$rc" = 4 ] || [ "$mine" = 0 ] || rm -f "$BUSY"
    55	        [ "$rc" = 0 ] && exit 0
    56	        [ "$rc" = 5 ] && { echo "flapmon $(up): rollback stale (rc 5, newer composite) -> exit" >> "$LOG"; exit 0; }
    57	        [ "$rc" = 3 ] && { echo "flapmon $(up): rollback STUCK (rc 3, it escalated itself) -> no retry" >> "$LOG"; exit 1; }
    58	        echo "flapmon $(up): rollback attempt $try failed rc=$rc" >> "$LOG"
    59	        [ "$try" -ge "${MK20_FLAP_TRIES:-3}" ] && break
    60	        try=$((try+1)); sleep "${MK20_FLAP_RETRY_S:-20}"
    61	      done
    62	      # still not stock: persist stock mode (flag file, read at boot) and hard-reset via the rescue watcher
    63	      echo "flapmon $(up): rollback failed $try times -> ${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite} + reboot" >> "$LOG"
    64	      touch "${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite}" && touch "${MK20_FLAP_REBOOT:-/mnt/SDCARD/cmd-reboot}"; sync
    65	      exit 1
    66	    fi
    67	  fi
    68	  prev=$s; sleep "${MK20_FLAP_POLL:-0.3}"
    69	done

     4	#   MK20_REL (verified tmpfs copy of this release), MK20_VERSION, MK20_HOME (/data/mk20).

    50	  # mandatory (1.1.0): it is the subnet-independent path, so a release that loses it must not be confirmed.
    51	  local f n; f=0
    52	  n=$(kd_count); [ "$n" = 1 ] && echo "PASS kd=1" || { echo "FAIL kd=$n"; f=1; }
    53	  [ -n "$(cat "$G/UDC" 2>/dev/null)" ] && echo "PASS udc bound" || { echo "FAIL udc unbound"; f=1; }

    56	  if composite_up; then

    58	    listening "${MK20_LL_IP:-169.254.7.2}:$BRIDGE_PORT" && echo "PASS bridge link-local" || { echo "FAIL bridge not on link-local"; f=1; }

   145	    log WARN boot "composite skipped: $why -> stock ACM+KD$([ $SADB = 1 ] && echo +adb)"
   146	    [ $SADB = 1 ] && stock_adb
   147	  else log ERROR boot "$why -> no USB changes (stock ACM+KD only)"; fi
   148	  sleep "${MK20_STABLE:-30}"
   149	  # a gadget failsafe (flap rollback + adb re-add, up to ~2.5 min) in progress is not a failure: wait it out (cap)
   150	  # The marker is set by flapmon BEFORE it launches a rollback and by cleanup(); a sample taken while (or just
   151	  # before) it appears is retaken after the failsafe ends. Total wait capped (stays well under the loader deadline).
   152	  w=0; busy_wait() { while [ -f "$TM/gadget.busy" ] && [ $w -lt "${MK20_BUSY_CAP:-150}" ]; do sleep 5; w=$((w+5)); done; }
   153	  busy_wait; try=0; src=1; grc=${grc:-0}
   154	  while [ $try -lt 3 ]; do
   155	    try=$((try+1)); r=$(selftest); src=$?
   156	    if [ -f "$TM/gadget.busy" ] && [ $w -lt "${MK20_BUSY_CAP:-150}" ]; then busy_wait; try=$((try-1)); continue; fi
   157	    [ $src = 0 ] && break; [ $try = 3 ] || sleep "${MK20_SELFTEST_GAP:-10}"
   158	  done
   159	  [ $w -gt 0 ] && log INFO selftest "waited ${w}s for gadget failsafe"
   160	  # On probation, a composite that failed its own setup (gadget rc!=0: ip/alias/bridge/bind -> die -> stock) is a
   161	  # release failure, not "degraded": roll back. Host-caused flap rollbacks happen only after rc 0 and stay WARN.
   162	  if [ $MODE = composite ] && [ "$grc" != 0 ] && [ "$(cat "$MK20_HOME/state/pending" 2>/dev/null)" = "$V" ]; then
   163	    r="$r
   164	FAIL composite setup rc=$grc during probation"; src=1; fi
   165	  echo "$r" > "$TM/selftest.txt"; log INFO selftest "rc=$src try=$try mode=$MODE $(echo "$r" | tr '\n' ';')"
   166	  # per-boot records: selftest-last.json + selftest-1..4.json (previous boots)
   167	  ST=$MK20_HOME/state; for k in 3 2 1; do [ -f "$ST/selftest-$k.json" ] && mv -f "$ST/selftest-$k.json" "$ST/selftest-$((k+1)).json"; done
   168	  [ -f "$ST/selftest-last.json" ] && mv -f "$ST/selftest-last.json" "$ST/selftest-1.json"
   169	  # decide on the 3-sample result ($src) BEFORE the (slower) health record, which can never change it
   170	  if [ $src = 0 ]; then
   171	    if [ -f "$TM/rescue-loader.sh" ]; then sh "$TM/rescue-loader.sh" confirm "$V" && log INFO confirm "ok" || log INFO confirm "not pending/refused"; fi
   172	  else log ERROR selftest "FAILED: not confirming; restoring stock ACM+KD+adb now"
   173	    if [ -f "$TM/gadget-boot.sh" ] && composite_up; then
   174	      # same contract as the confirm waiter: retry once (rc 3 = stuck, already escalated), then persist stock + reboot
   175	      rbn() { MK20_REL=$REL MK20_GADGET_LOG=$LOGDIR/gadget.log sh "$TM/gadget-boot.sh" --rollback-now >> "$LOGDIR/gadget.log" 2>&1 6>&- 8>&- 9>&-; }
   176	      rbn; rrc=$?
   177	      if [ $rrc != 0 ] && [ $rrc != 3 ]; then log WARN selftest "rollback rc=$rrc; retrying once"; sleep "${MK20_RB_RETRY_S:-20}"; rbn; rrc=$?; fi
   178	      if [ $rrc != 0 ] && [ $rrc != 3 ]; then log ERROR selftest "rollback failed twice (rc=$rrc) -> mk20-no-composite + reboot"
   179	        touch "${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite}" && touch "${MK20_FLAP_REBOOT:-/mnt/SDCARD/cmd-reboot}"; sync; fi
   180	    fi; fi
   181	  ( health ) > "$ST/selftest-last.json" 2>/dev/null
   182	  if [ "${MK20_MAINT:-1}" = 1 ]; then
   183	    export MK20_REL="$REL" MK20_VERSION="$V"
   184	    ( trap '' HUP INT; exec sh "$REL/sbin/mk20d" maint </dev/null >/dev/null 2>&1 6>&- 7>&- 8>&- 9>&- ) &
   185	  fi
   186	  ;;
   187	health) health;;
   188	maint)
   189	  exec 5>"$TM/mk20d-maint.lock"; flock -n 5 || exit 0   # single instance for the lifetime of the loop
   190	  log INFO maint "start pid=$$"
   191	  while :; do
   192	    cfg_load
   193	    log_rotate "$LOGDIR/mk20d.log"; log_rotate "$LOGDIR/gadget.log"
   194	    b=$TM/bridge.log; [ -f "$b" ] && [ "$(wc -c < "$b")" -gt 262144 ] && { tail -c 65536 "$b" > "$b.t" && cat "$b.t" > "$b"; rm -f "$b.t"; }   # O_APPEND writers keep appending
   195	    health > /dev/null 2>&1
   196	    sleep "${MK20_MAINT_S:-300}" 5>&-   # the lock dies with this loop, not with an orphaned sleep
   197	  done;;
   198	selftest) cfg_load; selftest; exit $?;;
   199	config) cfg_load; cfg_print;;
   200	version) echo "$V";;
   201	*) echo "usage: mk20d boot|health|selftest|config|maint|version"; exit 2;;
   202	esac
```
