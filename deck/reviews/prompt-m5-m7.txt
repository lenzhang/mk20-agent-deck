+#!/bin/sh
+# MK20 flap monitor (1.1.2; was an inline heredoc in gadget.sh). Started detached by gadget.sh:
+#   flapmon.sh <udc-state-file> <window_s> <drops> <log> <stopfile> -- <rollback command...>
+# Counts configured -> (not configured) link drops; >= <drops> within <window_s> -> touch the gadget.busy marker and
+# run the rollback command once, then exit. A host USB SUSPEND (configured -> suspended, e.g. Mac sleep) is not a
+# flap: it is logged but not counted, and the resume path (suspended -> ... -> configured) is not counted either.
+# Test env: MK20_UPTIME_CMD (seconds), MK20_FLAP_POLL (default 0.3), MK20_BUSY (marker path).
+U=$1; W=$2; N=$3; LOG=$4; STOP=$5; shift 5; [ "${1:-}" = -- ] && shift
+BUSY=${MK20_BUSY:-/tmp/mk20/gadget.busy}
+up() { if [ -n "${MK20_UPTIME_CMD:-}" ]; then $MK20_UPTIME_CMD; else cut -d. -f1 /proc/uptime; fi; }
+prev=$(cat "$U" 2>/dev/null); times=""
+while [ ! -f "$STOP" ]; do
+  s=$(cat "$U" 2>/dev/null)
+  if [ "$prev" = configured ] && [ "$s" = suspended ]; then
+    echo "flapmon $(up): host suspend (not counted)" >> "$LOG"
+  elif [ "$prev" = configured ] && [ "$s" != configured ]; then
+    now=$(up); nt=""; cnt=0
+    for t in $times $now; do [ $((now-t)) -le "$W" ] && { nt="$nt $t"; cnt=$((cnt+1)); }; done; times=$nt
+    echo "flapmon $now: link drop ($s), $cnt in ${W}s" >> "$LOG"
+    if [ "$cnt" -ge "$N" ]; then
+      echo "flapmon $now: FLAPPING -> rollback" >> "$LOG"; touch "$BUSY"; "$@"; exit 0
+    fi
+  elif [ "$prev" = suspended ] && [ "$s" = configured ]; then
+    echo "flapmon $(up): host resume" >> "$LOG"
+  fi
+  prev=$s; sleep "${MK20_FLAP_POLL:-0.3}"
+done

+  bound to the MK20 interface); Protocol A getInfo over TCP (deviceVersion); HID: Mac sees the 1d6b:0104 keyboard
+  and a Left-Shift tap written to /dev/hidg0 is consumed by the host within 3 s; Mac route to the device not via a

+def adb_cmd() -> list[str]:
+    a = os.environ.get("MK20_ADB")

+def sh(cmd: str, timeout: float = 20) -> tuple[int, str]:
+    try:
+        r = subprocess.run(adb_cmd() + ["shell", cmd], capture_output=True, text=True, timeout=timeout)
+        return r.returncode, r.stdout
+    except subprocess.TimeoutExpired:
+        return 124, ""

+class Run:
+    def __init__(self, quiet: bool = False):
+        self.results: list[dict] = []
+        self.quiet = quiet
+
+    def check(self, name: str, ok: bool, detail: str = "") -> bool:
+        self.results.append({"check": name, "pass": bool(ok), "detail": detail})
+        if not self.quiet:
+            print(f"{'PASS' if ok else 'FAIL'} {name}{(' - ' + detail) if detail else ''}", flush=True)
+        return ok
+
+    @property
+    def ok(self) -> bool:
+        return all(r["pass"] for r in self.results)

+def loader_status() -> dict:
+    _, out = sh("sh /data/mk20/rescue/loader.sh status")
+    d = dict(re.findall(r"(\w+)=(\S*)", out))
+    m = re.match(r"current=\S+ \d+ (\S+)", out)
+    d["current_v"] = m.group(1) if m else ""
+    d["raw"] = out.strip()
+    return d
+

+def health(v: str) -> dict:
+    _, out = sh(f"/data/mk20/releases/{v}/sbin/mk20d health", timeout=40)
+    try:

+    except Exception:  # noqa: BLE001

+def wait_adb(present: bool, limit: float) -> bool:
+    end = time.time() + limit
+    while time.time() < end:
+        rc, out = sh("echo up", timeout=5)
+        if (rc == 0 and "up" in out) == present:
+            return True
+        time.sleep(3)
+    return False
+

+def do_reboot(run: Run) -> None:

+# MK20 Agent firmware: recovery
+
+Work from the top of this list down, and stop at the first level that works. Levels 1–3 need only adb, or a TF
+card reader if adb is gone. Level 5 needs the TF card in the Mac. Nothing here writes the rootfs (p5).
+(The P0 GD32/VIA backup and restore manual is `deck/RECOVERY.md`.)
+
+`$MK20_ADB` is `~/mk20-backup/mkadb.sh` (MK20 only, never the watch). "Reboot" always means one of:
+- `$MK20_ADB shell 'touch /mnt/SDCARD/cmd-reboot && sync'`: the watcher runs a hardreset through the WDT, about
+  30 s later; or
+- a USB power cycle (unplug and replug), which needs a human.
+
+Never use `reboot`: the soft reboot hangs on this board.
+
+## 0. Look first (read-only)
+
+```sh
+$MK20_ADB shell 'sh /data/mk20/rescue/loader.sh status'
+$MK20_ADB shell 'v=$(cat /data/mk20/state/good); /data/mk20/releases/$v/sbin/mk20d health'
+$MK20_ADB shell 'tail -n 30 /data/mk20/logs/mk20d.log /mnt/SDCARD/mk20/logs/loader.log /mnt/SDCARD/mk20/logs/watcher.log'
+$MK20_ADB shell 'cat /tmp/mk20/stub.status /data/mk20/state/hardreset.last /mnt/SDCARD/gadget-ecm-hid.status'
+python3 tools/mk20_host_net.py status       # Mac side: service, routes, device subnet, ping
+python3 tools/mk20_accept.py                # full read-only check
+```
+
+Symptoms and where to go:
+
+| symptom | likely cause | go to |
+|---|---|---|
+| ping to 172.30.250.2 fails, adb works | Mac/device subnet mismatch (device rolled back to 1.0.x) | `mk20_host_net.py sync` |
+| black KD screens, USB present | KD lost ttyGS0 (composite half-down) | 1 (`mk20-no-composite`) |
+| USB keeps re-enumerating | flap; flapmon rolls back to stock after 5 drops in 60 s (Mac sleep/suspend is not counted, 1.1.2+) | 0, then 1 |
+| `good=` empty / `stub: … mismatch` | release or rescue files corrupt | 2 / 3 |
+| nothing enumerates after a power cycle | boot image or kernel broken | 5 |
+
+## 1. Flag files (FAT `/mnt/SDCARD/`, read at boot; then reboot)
+
+| flag | effect |
+|---|---|
+| `mk20-no-composite` | release runs in stock mode: ACM + KD (+adb), no ECM/HID/bridge |
+| `mk20-gadget-nohid` | composite without HID (ECM + ADB + bridge); use it on the ep3-only kernel |
+| `mk20-no-adb` | no adb in any mode |
+| `mk20-no-release` | loader skips releases: legacy `boot-every/` jobs only (stock + adb keeper) |
+| `mk20-no-watch` | boot stub does nothing at all: pure vendor behaviour, no rescue watcher, no adb |
+
+Over adb: `$MK20_ADB shell 'touch /mnt/SDCARD/mk20-no-composite && sync'`, then reboot. Without adb, put the TF
+card in the Mac (the FAT volume mounts in Finder), create the empty file, eject cleanly, and put the card back.
+Delete the flag to undo.
+
+Commands the watcher polls for every 2 s:
+- `cmd-reboot`: hardreset.
+- `cmd-rollback`: switch to the other A/B slot, then hardreset.
+- `cmd-ping`: replies "pong" on ttyGS0.
+- `job.sh` with `job.sh.sha256`: verified one-shot job, 300 s cap.
+
+## 2. Roll back the release
+
+- **Automatic:** a pending release that fails its self-test, fails composite setup during probation (1.1.1+), or
+  isn't confirmed within 600 s rolls back to `good` with at most one hard reset. Proven on hardware with 1.0.3
+  rolling back to 1.0.2.
+- **Manual:** `$MK20_ADB shell 'touch /mnt/SDCARD/cmd-rollback && sync'`. The watcher runs `loader rollback` and
+  then a hardreset. Over a shell, `sh /data/mk20/rescue/loader.sh rollback`, then reboot.
+- Afterwards, align the Mac: `python3 tools/mk20_host_net.py sync`. In a mismatch, the bound link-local path to
+  169.254.7.2 still works.
+
+## 3. Uninstall the firmware, or remove the stub
+
+- `$MK20_ADB shell 'sh /data/mk20/rescue/loader.sh uninstall'`, then reboot. This moves the slots and state aside
+  (`/data/mk20/uninstalled-<ts>/`) and parks the legacy composite job. The next boot is stock ACM+KD+adb with the
+  rescue watcher.
+- **Remove the boot stub** (pure vendor, nothing of ours runs): delete `/mnt/SDCARD/lunch.sh`, over adb or on the
+  Mac with a TF reader, or create `mk20-no-watch`. The vendor `qt_app1` only runs the stub if the file exists.
+- The rescue copies are checksum-verified. The stub tries `/data/mk20/rescue` first and falls back to
+  `/mnt/SDCARD/mk20`. A checksum *mismatch* runs nothing; fix it with `tools/mk20-rescue-update.sh`.
+
+## 4. overlayDir: write any file at the next boot (TF reader, no adb needed)
+
+The vendor `/etc/init.d/qt_app1` copies `/mnt/SDCARD/overlayDir/*` onto `/` at **every** boot, then empties the
+directory. This happens before `lunch.sh` runs.
+
+With the TF card in the Mac, make the target path under `overlayDir/`. For example, to restore a vendor file from
+backup: `overlayDir/data/appLunch.sh`, or `overlayDir/data/mk20/rescue/watcher.sh` together with its `.sha256`.
+Eject and boot. The copy is a single shot and `overlayDir` is empty again afterwards.
+
+- Vendor originals for comparison: the hashes in `share/vendor.sha256`; the backups are in `~/mk20-backup/`.
+- Security note: anyone with the TF card gets root this way. That is inherent to the vendor image. Keep physical
+  custody of the card.
+
+## 5. Kernel (boot image, p4) restore from the Mac
+
+Kernel images. All are 6451200 B and all report `uname` #221, so identify them **by sha only**:
+
+| name | p4 sha256 | features | Mac copy | device copy |
+|---|---|---|---|---|
+| stock | `9ac68ed98a70a3b5211dc8a25cf9427afdc54543ebfe003337c77032929029a8` | acm,adb | `~/mk20-backup/tf/live-221-parts/p4.img` | `/data/mk20/kernel/9ac68ed98a70/boot.img`, `/mnt/SDCARD/boot-stock.img` |
+| ep3 | `f2a19d3cb8fddc7f177cb518bb64e69645fadd3ce1b1422c4f056db39e4ee4c0` | acm,adb,ecm | `~/mk20-backup/tf/patched-ep3/p4-patched-ep3int.img` | `/data/mk20/kernel/f2a19d3cb8fd/boot.img` |
+| ep5out (running) | `7326d9e5ab9fd9f140b6cc7597b4eb0539b019139d12b52b4c133771e35843cc` | acm,adb,ecm,hid | `~/mk20-backup/tf/patched-ep5out/p4-patched-ep3int-ep5out.img` | `/data/mk20/kernel/7326d9e5ab9f/boot.img` |
+
+Steps:
+1. Power off and take the TF card out. Put it in the Mac's reader.
+2. Run `bash tools/tf-identify.sh` and note `diskN`. `tf-restore-boot.sh` refuses internal disks, anything not
+   3–5 GB, and an s4 that isn't exactly 6451200 B.
+3. Rehearse first: `bash tools/tf-restore-boot.sh --disk diskN --image ~/mk20-backup/tf/live-221-parts/p4.img
+   --slice s4 --dry-run`.
+4. Run the same command without `--dry-run`. It asks for sudo, writes slice **s4 only**, and reads back the sha.
+5. Eject, put the card back, and power on.
+   - **Stock kernel:** releases see an unknown or `acm,adb` p4 and make no composite. You get stock ACM+KD+adb.
+   - **Patched kernel:** re-flash `p4-patched-ep3int-ep5out.img` the same way.
+
+Whole card (last resort): `bash tools/tf-restore-boot.sh --disk diskN --whole-image <img> [--dry-run]`. Images:
+- the live #221 partitions, `~/mk20-backup/tf/live-221-parts/` (SHA256SUMS; p6 UDISK isn't in the backup, so
+  `/data` comes back empty);
+- the vendor image `MK20_V1.0.img`.
+
+Per-kernel notes are on the device in `/data/mk20/kernel/<sha12>/RECOVERY.txt` (source:
+`firmware/kernel-notes/`).
+
+## 6. Mac network (undo the M4 changes)
+
+See INSTALL.md §5. In short: `python3 tools/mk20_host_net.py linklocal-remove` removes the "MK20 link-local"
+service, and `revert --force` sets Serial back to 10.20.0.1/24. The service order before M4 is in
+`~/mk20-backup/netservices-before-m4.txt`. No other service was changed, and the mihomo config wasn't touched.

+# MK20 Agent firmware: install, upgrade, Mac setup
+
+All commands run on the Mac from `deck/` unless marked **device**. Every device command goes through adb aimed at
+the MK20 **only**:
+
+```sh
+export MK20_ADB=~/mk20-backup/mkadb.sh   # picks the MK20 transport; never the watch (WATCH-SERIAL / 192.168.88.103)
+```
+
+House rules:
+- Never `killall adbd` in a foreground adb shell.
+- Never soft `reboot` (it hangs on this board). Reboots go through the watcher (`cmd-reboot`, which triggers a
+  hardreset through the WDT).
+- Anything that changes USB runs detached, with a rollback that restores ACM+KD.
+
+## 1. Prerequisites (once)
+
+- Back up first:
+  - TF partitions: `~/mk20-backup/tf/live-221-parts/`, with sha values in NOTES-crash-20261018-and-tf-backup.md.
+  - `/data`: use `tar -X tools/backup-exclude.txt`. This keeps `/data/mk20/secrets/` (the VNC password) on the
+    device.
+- Kernel inventory on the device: `/data/mk20/kernel/<sha12>/{boot.img,MANIFEST,RECOVERY.txt}` plus `ALLOWLIST`
+  (`<p4 sha256> <name> <features>`). mk20d makes no USB changes on a p4 that isn't listed.
+- Host tools: Python 3 and `uv` (for pyserial), `adb`. For the test suite, the box with BusyBox 1.27.2 at
+  `/workspace/tools/bb127/busybox`.
+
+## 2. First install (from the ad-hoc v3.5 layout)
+
+1. **Rescue tier:** build first (`make release` creates `out/rescue/` with `.sha256` files), then:
+   ```sh
+   sh tools/mk20-rescue-update.sh out/rescue loader.sh hardreset.sh watcher.sh lunch.sh
+   ```
+   Each file goes to `/data/mk20/rescue/` and then `/mnt/SDCARD/mk20/`, verified before the atomic rename, so one
+   valid copy always exists. The FAT `lunch.sh` stub starts the first verified watcher at boot.
+2. **First release:** `sh tools/mk20-install.sh out/mk20-fw-<v>.tar --reboot --wait`.
+3. **Legacy jobs:** once a release launches, the watcher skips `boot-every/`. The old composite job is parked in
+   `/mnt/SDCARD/mk20/boot-every.disabled/`; `loader uninstall` also does this. `boot-every/adb.sh` stays as the
+   legacy-mode adb keeper.
+
+## 3. Upgrade (normal path)
+
+```sh
+shasum -a 256 -c ~/mk20-backup/releases/mk20-fw-<v>.tar.sha256
+sh tools/mk20-install.sh ~/mk20-backup/releases/mk20-fw-<v>.tar --reboot --wait
+```
+
+- The installer pins the tarball sha and pushes it to `/data/mk20/incoming/upload.tar`. It then runs the
+  verified rescue loader: `loader install`, which makes a private copy, checks the sha and free space, validates
+  the tar listing, extracts into `.incoming-<v>`, compares the file set with the MANIFEST, and atomically renames
+  to `releases/<v>`. Then `switch` marks the release **pending** and the loader prunes old releases.
+- `--reboot` drops `cmd-reboot` on the FAT; the watcher's hardreset reboots the board about 30 s later.
+- **Probation:** the loader boots the pending release, and mk20d runs a self-test (KD=1, UDC, ttyGS0, watchdog,
+  bridge on the ECM IP and 169.254.7.2, HID). A pass means `confirm`, so the release becomes good. Any of these
+  roll back automatically to the previous good release, with no host needed:
+  - the self-test fails;
+  - the composite fails its own setup during probation (1.1.1+);
+  - nothing confirms within 600 s.
+- If the installer refuses with "is pending", the previous install hasn't been booted yet. Reboot, or let
+  probation finish.
+
+Check:
+
+```sh
+$MK20_ADB shell 'sh /data/mk20/rescue/loader.sh status'          # current=.. good=<v> pending= ...
+$MK20_ADB shell '/data/mk20/releases/<v>/sbin/mk20d health'       # JSON, "status":"ok"
+$MK20_ADB shell 'cat /data/mk20/state/hardreset.last'             # fat_ro=y mount_hung=n (clean FAT remount)
+```
+
+Logs: `/data/mk20/logs/{mk20d,gadget}.log` (rotated), `/mnt/SDCARD/mk20/logs/{loader,watcher}.log`, and
+`/tmp/mk20/bridge.log`.
+
+Uninstall (back to stock ACM+KD+adb with the rescue watcher): **device** `sh /data/mk20/rescue/loader.sh
+uninstall`, then reboot through the watcher.

+ignored with a warning (`mk20d config` shows the effective values).

+|---|---|---|
+| `ECM_SUBNET` | `172.30.250.0/24` (1.1.0+) | a /24 network address; device .2, Mac .1 |

+## 5. Mac setup (M4, firmware 1.1.0+)
+
+The ECM link is the Mac network service whose hardware address is **02:11:22:33:44:56** (here: en13, service
+"Serial"). `tools/mk20_host_net.py` changes **only** that service. Every change is verified (`route get` must use
+the MK20 interface, and ping must answer) and restored in full on failure or Ctrl-C.
+
+```sh
+python3 tools/mk20_host_net.py status       # read-only
+python3 tools/mk20_host_net.py sync         # Mac side := device's active subnet (read over adb). Use after any up/downgrade
+python3 tools/mk20_host_net.py apply        # 172.30.250.1/24; refuses unless the device is confirmed on 172.30.250.0/24
+python3 tools/mk20_host_net.py linklocal    # add "MK20 link-local" = 169.254.7.1/16 on the same port
+```
+
+State on this Mac (since 19:01 SGT, 2026-10-18):
+- **Serial** (en13): manual 172.30.250.1 / 255.255.255.0, no router, so the default route is untouched.
+- **MK20 link-local** (en13, same hardware port): manual 169.254.7.1 / 255.255.0.0. This is a second service, at
+  position 5 in the service order. Link-local traffic for other networks still uses the primary interface (en0).
+- Service order before M4: `~/mk20-backup/netservices-before-m4.txt`.
+
+**Undo:**
+
+```sh
+python3 tools/mk20_host_net.py linklocal-remove      # removes the "MK20 link-local" service only
+python3 tools/mk20_host_net.py revert --force        # Serial back to 10.20.0.1/24 (the 1.0.x address)
+# by hand, equivalent:
+networksetup -removenetworkservice "MK20 link-local"
+networksetup -setmanual Serial 10.20.0.1 255.255.255.0
+```
+
+`revert` without `--force` refuses unless the device actually runs a 1.0.x subnet. On a 1.1.x device you'd lose the
+primary path, but the link-local path still works.
+
+**TUN (mihomo or any other):** mihomo's `tun.route-exclude-address` already includes 172.16.0.0/12 and
+10.0.0.0/8, so nothing needs changing. Check with `route -n get 172.30.250.2`, which should say `interface: en13`.
+The daemon and the host tools also pin their sockets to the MK20 interface (`IP_BOUND_IF`), so a TUN can't
+capture them even without the exclude.
+
+**Daemon and tools:** these follow the device automatically.
+- The daemon (`feature/p0-mvp`, `mk20deck/transport.py`) tries 172.30.250.2, then 10.20.0.2, then 169.254.7.2,
+  all bound to the interface.
+- `tools/tcp_transport.py connect_device()` tries 172.30.250.2, then 169.254.7.2.
+- `mk20ctl` over TCP:
+  `MK20_PORT=socket://172.30.250.2:17890 uv run --with pyserial python tools/mk20ctl-url.py info`.

+# MK20 Agent firmware: releases
+
+Applies to: Waveshare MK20 (Allwinner T113, Tina Linux 5.4.61 #221, BusyBox 1.27.2). The firmware is a userland
+layer on `/data` (ext4/UDISK). It never writes the rootfs. Kernels are only swapped as whole boot images (p4,
+see RECOVERY.md level 5).
+
+## What is a release
+
+A release lives in `/data/mk20/releases/<semver>/`:
+
+| path | what |
+|---|---|
+| `sbin/mk20d` | entry point: gates, gadget launch, self-test, confirm, `health`, `selftest`, `config`, `maint`, `version` |
+| `share/mk20lib.sh` | whitelisted config loader (`/data/mk20/config/mk20.conf`, never sourced) + structured logs/rotation |
+| `share/gadget.sh` | ECM + HID + ADB composite, PTY↔TCP bridge, rollback to stock ACM+KD |
+| `share/flapmon.sh` | flap monitor (1.1.2+): ≥5 link drops in 60 s → rollback; host USB suspend is not a flap |
+| `share/mk20-add-adb.sh` | stock-mode adb add (fallback) |
+| `share/vendor.sha256` | expected hashes of the 4 vendor files the KD shims depend on |
+| `bin/ttygs_tcp_bridge` | static ARM PTY↔TCP bridge (bound to `usb0`, explicit `--listen` only) |
+| `lib/usb_f_ecm.ko` | ECM function module for the #221 kernel |
+| `MANIFEST` | `name/version/build/kernels` header, `--`, then `sha256 mode path` for every file |
+
+The **rescue tier** (`watcher.sh`, `hardreset.sh`, `loader.sh`, `lunch.sh` stub) is *not* part of any release. It
+lives in `/data/mk20/rescue/` and `/mnt/SDCARD/mk20/`, each file with a `.sha256` next to it. It is updated
+separately with `tools/mk20-rescue-update.sh`.
+
+## Versioning
+
+- `MAJOR.MINOR.PATCH`, each part 0–999 (`firmware/VERSION`, one line).
+- PATCH: fixes with no behaviour change visible to the host. MINOR: host-visible change, such as the default
+  subnet, new knobs or new self-test gates. MAJOR: a layout or rescue-protocol change that needs a newer loader.
+- Never rebuild a released version with different content. Bump PATCH instead. The installer reuses an identical
+  installed version and refuses a pending one.
+- `firmware/KERNELS` lists the p4 sha256 values a release supports. These are copied into the MANIFEST
+  `kernels=` line. At run time, mk20d also needs `/data/mk20/kernel/ALLOWLIST` to name the running p4 and its
+  features.
+
+## Build
+
+```sh
+cd deck
+make test                                   # shellcheck + box tests (BusyBox 1.27.2) + host tests: must be green
+sh tools/mk20-release-build.sh firmware out <build-id>   # or: make release (build id = git short sha)
+# -> out/mk20-fw-<v>/ , out/mk20-fw-<v>.tar , out/mk20-fw-<v>.tar.sha256 , out/rescue/ (+ .sha256)
+```
+
+The build needs GNU tar and GNU stat (the box, or any Linux). It is reproducible: same tree and build id give the
+same tarball sha. It rejects symlinks, unsafe names and a bad VERSION.
+
+## Release checklist
+
+1. `make test` is green on the box.
+2. Code review of the milestone diff: check the mihomo AI route first (US-Exit-Hy2), then the reviews in
+   `deck/reviews/`. Fix every BLOCKER and MAJOR before building.
+3. Build. Copy the `.tar` and `.tar.sha256` to the Mac (`~/mk20-backup/releases/`) and check them with
+   `shasum -a 256 -c`.
+4. Install on hardware (INSTALL.md §3) with `--reboot`. Watch for `good=<v>` in `loader status` and a
+   `mk20d health` status of `ok`.
+5. Run the acceptance script (`tools/mk20_accept.py`, INSTALL.md §6).
+6. Commit and push: the code, the review files, and the NOTES entry with the hardware evidence.
