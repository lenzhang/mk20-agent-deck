   139	# Transition mutex (1.1.4, r5): ONE flock (fd 7 on /tmp/mk20/transition.lock) is held for the WHOLE setup or the WHOLE
   140	# rollback, so they can never overlap (flapmon, the confirm waiter and mk20d may all ask for --rollback-now). Every
   141	# process spawned while it is held gets 7>&- (adbd, bridge loop, KD, add-adb, flapmon, confirm waiter), so only the
   142	# owner holds it and it is released when the owner exits. A rollback that finds stock verified is a no-op: exit 0
   143	# always means stock is up. Every lock step is checked (fail closed: no mutation without the mutex).
   144	# Owner-local bounded supervisor: the rollback runs cleanup in its own child and waits at most MK20_RB_TIMEOUT
   145	# (default 330 s); on timeout it stops, kills and reaps ITS OWN child tree before releasing the mutex. No caller ever
   146	# kills another owner; a waiter that cannot get the mutex in MK20_RB_LOCK_TRIES*0.25 s (default 900 s) gives up (rc 4).
   147	# >>> rollback-block (tests extract from here to the matching end marker)

   149	stock_ok() { [ -e $C/acm.usb0 ] && [ ! -e $C/ecm.usb0 ] && [ ! -e $C/hid.usb0 ] && [ -n "$(udc_now)" ] &&
   150	  [ -c /dev/ttyGS0 ] && [ ! -L /dev/ttyGS0 ] && ! grep -qE ' /data/(setusbconfig|appLunch\.sh) ' /proc/mounts &&

   152	rb_status() { echo "$1" > /mnt/SDCARD/gadget-ecm-hid.status 2>/dev/null; }
   153	pstart() { awk '$3 != "Z" {print $22}' "/proc/$1/stat" 2>/dev/null; }   # empty if gone or a zombie
   154	# tl_lock <tries>: take the transition mutex on fd 7 (tries 0.25 s apart; BusyBox flock has no -w). 1 = not held.
   155	tl_lock() {
   156	  mkdir -p "$(dirname "$TL")" 2>/dev/null || return 1
   157	  touch "$TL" 2>/dev/null || return 1   # an exec redirect error would exit the shell (special builtin)
   158	  exec 7>"$TL" || return 1
   159	  n=0; until flock -n 7; do n=$((n+1)); [ "$n" -ge "$1" ] && { exec 7>&-; return 1; }; sleep 0.25; done
   160	}
   161	kids() { for d in /proc/[0-9]*; do [ "$(awk '/^PPid:/{print $2}' "$d/status" 2>/dev/null)" = "$1" ] && echo "${d#/proc/}"; done; }
   162	# kill_tree prints "pid:starttime" for every process it stops (taken while STOPped, before SIGKILL reparents the
   163	# descendants to init) so bounded_wait can check the WHOLE tree afterwards (r6: an unkillable grandchild may hold fd 7).
   164	kill_tree() { kill -STOP "$1" 2>/dev/null; echo "$1:$(pstart "$1")"; for k in $(kids "$1"); do kill_tree "$k"; done; kill -9 "$1" 2>/dev/null; }
   165	# any_alive "<pid:start ...>": 0 if any listed process still runs with the same start time (not a zombie)
   166	any_alive() { for e in $1; do [ -n "${e#*:}" ] && [ "$(pstart "${e%%:*}")" = "${e#*:}" ] && return 0; done; return 1; }
   167	# uint_cap <value> <default> <min> <max>: a validated, clamped decimal (no octal, no overflow: >6 digits = max).
   168	# Every timeout/lock-try knob goes through this BEFORE any lock or fork (r5: bad arithmetic must not kill the owner).
   169	uint_cap() {
   170	  case "$1" in ''|*[!0-9]*) echo "$2"; return;; esac
   171	  v=${1#"${1%%[!0]*}"}; [ -n "$v" ] || v=0
   172	  [ ${#v} -gt 6 ] && { echo "$4"; return; }
   173	  [ "$v" -lt "$3" ] && v=$3; [ "$v" -gt "$4" ] && v=$4; echo "$v"
   174	}
   175	# r5 #5: cleanup worst case on a HEALTHY board = kd_stop<=32 + umount<=10 + kd_start<=23 + readd_adb (g1 wait<=60,
   176	# 5, add-adb ~4 x7, 6x15 re-add window) ~= 260 s -> 330 s so a healthy rollback incl. add-adb is never cut off.
   177	RBRETRY=$(uint_cap "${MK20_RB_RETRY_S:-}" 20 0 300)       # queued/manual rollback: pause before its one retry (s)
   178	RBT=$(uint_cap "${MK20_RB_TIMEOUT:-}" 330 1 900)          # rollback / post-timeout cleanup bound (s)
   179	RBTRIES=$(uint_cap "${MK20_RB_LOCK_TRIES:-}" 3600 1 4800)  # rollback mutex wait (x0.25 s) >= SUT+RBT+30 s owner hold
   180	SUTRIES=$(uint_cap "${MK20_SETUP_LOCK_TRIES:-}" 40 1 4800) # setup mutex wait (x0.25 s)
   181	SUT=$(uint_cap "${MK20_SETUP_TIMEOUT:-}" 450 1 1800)       # whole setup incl. its own failsafe (~60 s + cleanup <=330 s)
   182	# bounded_wait <pid> <s>: 0 = the child exited in time (rc in BW_RC); 1 = timed out: its whole tree was stopped,
   183	# SIGKILLed and the child reaped; 2 = timed out AND still alive 10 s after SIGKILL (D state, e.g. a stuck UDC write):
   184	# not waited for (an unbounded wait would wedge us). "Alive" covers EVERY process of the tree snapshotted by
   185	# kill_tree, not just the root (r6). Only ever used on OUR OWN children.
   186	bounded_wait() {
   187	  bst=$(pstart "$1"); bi=0; BW_RC=1
   188	  while [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ] && [ $bi -lt $(($2*4)) ]; do sleep 0.25; bi=$((bi+1)); done
   189	  if [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ]; then
   190	    bsnap="$1:$bst $(kill_tree "$1")"; bi=0
   191	    while any_alive "$bsnap" && [ $bi -lt 40 ]; do sleep 0.25; bi=$((bi+1)); done
   192	    any_alive "$bsnap" && return 2
   193	    wait "$1" 2>/dev/null; return 1
   194	  fi
   195	  wait "$1" 2>/dev/null; BW_RC=$?; return 0
   196	}

   198	# stock mode (flag read at boot) and hard-reset via the rescue watcher (cmd-reboot) - same path as flapmon's.
   199	# r6: the writes run in a bounded child WITHOUT fd 7 (a hung FAT write cannot wedge the mutex); success = both files
   200	# exist afterwards (sync cannot mask a failed touch). 0 = escalated, 1 = could not persist (caller exits 1, not 3).
   201	escalate() {

   206	  [ "$ebw" = 0 ] && [ "$BW_RC" = 0 ] && [ -e "$EF" ] && [ -e "$ER" ] && return 0
   207	  log "ESCALATE FAILED (bw=$ebw rc=$BW_RC): $EF / $ER not persisted"; return 1
   208	}

   213	# owner is mid-transition and owns gadget.busy) | 5 stale (r5 #7: MK20_GEN names an older composite than the current

   238	    rb_status rollback-NOT-verified

   240	    [ $a -ge $RBA ] && break
   241	    sleep "$RBRETRY"

=== deck/firmware/release/share/flapmon.sh
     1	#!/bin/sh
     2	# MK20 flap monitor (1.1.2+; was an inline heredoc in gadget.sh). Started detached by gadget.sh:
     3	#   flapmon.sh <udc-state-file> <window_s> <drops> <log> <stopfile> -- <rollback command...>
     4	# Counts link drops; >= <drops> within <window_s> -> touch the gadget.busy marker and run the rollback command once,
     5	# then exit. A drop is: configured -> anything but configured/suspended, or a SHORT suspend (< MK20_FLAP_SUSP_S,
     6	# default 5 s) that ends in anything but configured (a bus reset/glitch the poll caught as idle). A long suspend
     7	# (Mac sleep) is not a flap: suspend, resume and a resume reset / unplug after it are logged, not counted.
     8	# Single instance (flock on fd 8, not inherited by the rollback). The stopfile is re-checked right before rollback.
     9	# The busy marker is cleared by flapmon only if it created it (sticky across retries; kept after a final rc 4). A failed rollback is retried (MK20_FLAP_TRIES, default 3,
    10	# MK20_FLAP_RETRY_S apart); if all fail, the persistent mk20-no-composite flag is set and the watcher hard-resets
    11	# (cmd-reboot), so the next boot runs stock until the flag is deleted.
    12	# The rollback command bounds itself (gadget.sh --rollback-now: transition mutex + owner-local supervisor that kills
    13	# and reaps its own cleanup tree after MK20_RB_TIMEOUT), so a hung rollback returns rc!=0 here and is retried/escalated.
    14	# Writes the ready marker (MK20_FLAP_READY, default /tmp/mk20/flapmon.ready) once the lock is held.
    15	# Test env: MK20_UPTIME_CMD (seconds), MK20_FLAP_POLL (default 0.3), MK20_BUSY, MK20_FLAP_LOCK, MK20_FLAP_SUSP_S.
    16	[ $# -ge 7 ] && [ "$6" = -- ] || { echo "usage: flapmon.sh state window drops log stopfile -- cmd..." >&2; exit 2; }
    17	U=$1; W=$2; N=$3; LOG=$4; STOP=$5; shift 6
    18	case "$W$N" in ''|*[!0-9]*) echo "flapmon: window/drops must be integers" >&2; exit 2;; esac
    19	BUSY=${MK20_BUSY:-/tmp/mk20/gadget.busy}; SUSP=${MK20_FLAP_SUSP_S:-5}
    20	LOCK=${MK20_FLAP_LOCK:-/tmp/mk20/flapmon.lock}; mkdir -p "$(dirname "$LOCK")"
    21	exec 8>"$LOCK"; i=0; while ! flock -n 8; do i=$((i+1)); [ $i -lt 10 ] && { sleep 0.5; continue; }
    22	  { echo "flapmon $(cut -d. -f1 /proc/uptime): another monitor holds $LOCK" >> "$LOG"; exit 3; }; done
    23	READY=${MK20_FLAP_READY:-/tmp/mk20/flapmon.ready}; mkdir -p "$(dirname "$READY")"; : > "$READY"   # lock held: gadget.sh waits for this
    24	up() { if [ -n "${MK20_UPTIME_CMD:-}" ]; then $MK20_UPTIME_CMD; else cut -d. -f1 /proc/uptime; fi; }
    25	prev=$(cat "$U" 2>/dev/null); times=""; sus=""
    26	while [ ! -f "$STOP" ]; do
    27	  s=$(cat "$U" 2>/dev/null); now=""
    28	  if [ "$s" != "$prev" ]; then
    29	    if [ "$prev" = configured ] && [ "$s" = suspended ]; then
    30	      sus=$(up); echo "flapmon $sus: host suspend" >> "$LOG"
    31	    elif [ "$prev" = configured ]; then
    32	      now=$(up)
    33	    elif [ "$prev" = suspended ]; then
    34	      t=$(up); d=$((t - ${sus:-$t}))
    35	      if [ "$s" = configured ]; then echo "flapmon $t: host resume after ${d}s" >> "$LOG"
    36	      elif [ -n "$sus" ] && [ "$d" -lt "$SUSP" ]; then now=$t
    37	      else echo "flapmon $t: '$s' after ${d}s suspend (resume reset/unplug, not counted)" >> "$LOG"; fi
    38	      sus=""
    39	    fi
    40	  fi
    41	  if [ -n "$now" ]; then
    42	    nt=""; cnt=0
    43	    for t in $times $now; do [ $((now-t)) -le "$W" ] && { nt="$nt $t"; cnt=$((cnt+1)); }; done; times=$nt
    44	    echo "flapmon $now: link drop ($s), $cnt in ${W}s" >> "$LOG"
    45	    if [ "$cnt" -ge "$N" ]; then
    46	      [ -f "$STOP" ] && exit 0
    47	      echo "flapmon $now: FLAPPING -> rollback" >> "$LOG"
    48	      try=1; mine=0
    49	      while :; do
    50	        # r5 #6: only clear a marker we created, and never after a refusal (rc 4: another owner holds the mutex and
    51	        # owns gadget.busy; it clears it itself under the mutex). rc 5 = stale generation: our composite is gone.
    52	        # GPT r7: ownership is sticky across retries, so a marker we created is removed on every terminal path.
    53	        [ "$mine" = 1 ] || [ -e "$BUSY" ] || { touch "$BUSY"; mine=1; }
    54	        "$@" 8>&-; rc=$?
    55	        [ "$rc" = 4 ] || [ "$mine" = 0 ] || { rm -f "$BUSY"; mine=0; }
    56	        [ "$rc" = 0 ] && exit 0
    57	        [ "$rc" = 5 ] && { echo "flapmon $(up): rollback stale (rc 5, newer composite) -> exit" >> "$LOG"; exit 0; }
    58	        [ "$rc" = 3 ] && { echo "flapmon $(up): rollback STUCK (rc 3, it escalated itself) -> no retry" >> "$LOG"; exit 1; }
    59	        echo "flapmon $(up): rollback attempt $try failed rc=$rc" >> "$LOG"
    60	        [ "$try" -ge "${MK20_FLAP_TRIES:-3}" ] && break
    61	        try=$((try+1)); sleep "${MK20_FLAP_RETRY_S:-20}"
    62	      done
    63	      [ "$mine" = 1 ] && [ "$rc" != 4 ] && rm -f "$BUSY"   # after a final refusal the owner may still hold it (mk20d caps its wait)
    64	      # still not stock: persist stock mode (flag file, read at boot) and hard-reset via the rescue watcher
    65	      echo "flapmon $(up): rollback failed $try times -> ${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite} + reboot" >> "$LOG"
    66	      touch "${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite}" && touch "${MK20_FLAP_REBOOT:-/mnt/SDCARD/cmd-reboot}"; sync
    67	      exit 1
    68	    fi
    69	  fi
    70	  prev=$s; sleep "${MK20_FLAP_POLL:-0.3}"
    71	done

    32	  while read -r h f || [ -n "$h" ]; do
    33	    [ -n "$h" ] || continue
    34	    n=$((n+1))
    35	    shim_mounted "$f" && continue
    36	    [ "$(sha256sum "$R$f" 2>/dev/null | cut -d' ' -f1)" = "$h" ] || { echo "mismatch:$f"; return 1; }
    37	  done < "$REL/share/vendor.sha256"
    38	  [ "$n" -ge 4 ] || { echo "vendor list incomplete ($n)"; return 1; }
    39	  echo ok
    40	}
    41	num() { case "$1" in ''|*[!0-9]*) echo 0;; *) echo "$1";; esac; }
    42	kd_count() { pgrep -f '^/data/KeyboardDevice' | wc -l | tr -d ' '; }   # not -x: BusyBox 1.27 -x never matches KD (1.0.3 field failure)
    43	composite_up() { [ -e "$C/ecm.usb0" ]; }
    44	listening() { netstat -ltn 2>/dev/null | awk -v a="$1" '$4==a {ok=1} END{exit !ok}'; }   # exact local address
    45	lock_held() { ( exec 5>"$1"; ! flock -n 5 ) 2>/dev/null; }   # a live instance holds the lock
    46	wdt_armed() { ${MK20_UBUS:-ubus} call system watchdog 2>/dev/null | grep -q '"running"'; }

    48	selftest() { # FAIL = host-independent breakage. Composite rollback to stock is "degraded" (WARN): flapmon rolls back on
    49	  # host-side USB flaps, and stock ACM+KD(+adb) stays reachable. With the composite up, the link-local rescue listener is
    50	  # mandatory (1.1.0): it is the subnet-independent path, so a release that loses it must not be confirmed.
    51	  local f n; f=0
    52	  n=$(kd_count); [ "$n" = 1 ] && echo "PASS kd=1" || { echo "FAIL kd=$n"; f=1; }
    53	  [ -n "$(cat "$G/UDC" 2>/dev/null)" ] && echo "PASS udc bound" || { echo "FAIL udc unbound"; f=1; }
    54	  [ -e "$R/dev/ttyGS0" ] && echo "PASS ttyGS0" || { echo "FAIL no ttyGS0"; f=1; }
    55	  wdt_armed && echo "PASS watchdog armed" || { echo "FAIL watchdog not armed"; f=1; }
    56	  if composite_up; then
    57	    listening "$ECM_IP:$BRIDGE_PORT" && echo "PASS bridge $ECM_IP:$BRIDGE_PORT" || { echo "FAIL bridge not on $ECM_IP:$BRIDGE_PORT"; f=1; }
    58	    listening "${MK20_LL_IP:-169.254.7.2}:$BRIDGE_PORT" && echo "PASS bridge link-local" || { echo "FAIL bridge not on link-local"; f=1; }
    59	    listening "0.0.0.0:$BRIDGE_PORT" && { echo "FAIL bridge on 0.0.0.0"; f=1; }
    60	    if [ -e "$C/hid.usb0" ]; then [ -e "$R/dev/hidg0" ] && echo "PASS hid" || { echo "FAIL no hidg0"; f=1; }; fi
    61	  else echo "WARN composite down (stock ACM+KD$([ -e "$C/ffs.adb" ] && echo +adb))"
    62	    for n in /data/appLunch.sh /data/setusbconfig; do shim_mounted "$n" && { echo "FAIL shim still mounted on $n with composite down"; f=1; }; done; fi
    63	  n=$(df -Pk "$S" 2>/dev/null | awk 'END{print $4}'); case "$n" in ''|*[!0-9]*) ;; *) [ "$n" -lt 1024 ] && echo "WARN FAT free ${n} KiB";; esac
    64	  return $f
    65	}

    67	jstr() { # JSON string: drop control bytes, escape backslash and quote
    68	  printf '"%s"' "$(printf '%s' "$1" | tr '\n\t' '  ' | LC_ALL=C tr -cd ' -~' | sed 's/\\/\\\\/g; s/"/\\"/g')"; }
    69	jbool() { if "$@" >/dev/null 2>&1; then echo true; else echo false; fi; }
    70	health() {
    71	  local kl up fat dat mem links l st rc mounts
    72	  cfg_load
    73	  kl=$(kernel_line); up=$(cut -d. -f1 /proc/uptime)
    74	  fat=$(df -Pk "$S" 2>/dev/null | awk 'END{print $4}'); dat=$(df -Pk "$MK20_HOME" 2>/dev/null | awk 'END{print $4}')
    75	  mem=$(awk '/MemAvailable/{print $2}' /proc/meminfo)
    76	  links=""; for l in "$C"/*usb0 "$C"/*adb; do [ -e "$l" ] && links="$links${l##*/} "; done
    77	  st=$(selftest 2>/dev/null); rc=$?
    78	  mounts=0; for l in /data/appLunch.sh /data/setusbconfig; do shim_mounted "$l" && mounts=$((mounts+1)); done
    79	  {
    80	    printf '{'
    81	    printf '"version":%s,"uptime_s":%s,' "$(jstr "$V")" "$(num "$up")"
    82	    printf '"kernel":{"p4":%s,"name":%s,"features":%s},' "$(jstr "$(p4sha)")" "$(jstr "$(echo "$kl" | cut -d' ' -f2)")" "$(jstr "$(echo "$kl" | cut -d' ' -f3)")"
    83	    printf '"vendor":%s,' "$(jstr "$(vendor_check)")"
    84	    printf '"usb":{"udc":%s,"state":%s,"functions":%s,"composite":%s},' "$(jstr "$(cat "$G/UDC" 2>/dev/null)")" \
    85	      "$(jstr "$(cat "$R"/sys/class/udc/*/state 2>/dev/null | head -n 1)")" "$(jstr "$links")" "$(jbool composite_up)"
    86	    printf '"net":{"usb0":%s,"mtu":%s,"subnet":%s,"bridge":%s,"hidg0_writable":%s},' "$(jstr "$(ifconfig usb0 2>/dev/null | sed -n 's/.*inet addr:\([0-9.]*\).*/\1/p')")" \
    87	      "$(num "$(cat "$R/sys/class/net/usb0/mtu" 2>/dev/null)")" "$(jstr "$ECM_SUBNET")" "$(jbool listening "$ECM_IP:$BRIDGE_PORT")" "$(jbool test -w "$R/dev/hidg0")"
    88	    printf '"kd":%s,"adbd":%s,"flapmon":%s,"maint":%s,"shim_mounts":%s,' "$(num "$(kd_count)")" "$(jbool pidof adbd)" \
    89	      "$(jbool pgrep -f mk20-flapmon.sh)" "$(jbool lock_held "$TM/mk20d-maint.lock")" "$mounts"
    90	    printf '"watchdog":%s,"mem_avail_kb":%s,"fat_free_kb":%s,"data_free_kb":%s,' "$(jbool wdt_armed)" "$(num "$mem")" "$(num "$fat")" "$(num "$dat")"
    91	    printf '"release":{"good":%s,"pending":%s,"last_rollback":%s},' "$(jstr "$(head -n 1 "$MK20_HOME/state/good" 2>/dev/null)")" \
    92	      "$(jstr "$(head -n 1 "$MK20_HOME/state/pending" 2>/dev/null)")" "$(jstr "$(head -n 1 "$MK20_HOME/state/last-rollback" 2>/dev/null)")"
    93	    printf '"config":{"src":%s,"warnings":%s},' "$(jstr "$CFG_SRC")" "$(jstr "$CFG_WARN")"
    94	    printf '"selftest":{"pass":%s,"detail":%s},' "$([ "$rc" = 0 ] && echo true || echo false)" "$(jstr "$st")"
    95	    printf '"status":%s' "$(jstr "$(if [ "$rc" != 0 ]; then echo fail; elif echo "$st" | grep -q '^WARN'; then echo degraded; else echo ok; fi)")"
    96	    printf '}\n'
    97	  } > "$TM/health.json.tmp.$$" && mv -f "$TM/health.json.tmp.$$" "$TM/health.json" || { rm -f "$TM/health.json" "$TM/health.json.tmp.$$"; return 1; }
    98	  cat "$TM/health.json"
    99	}

   101	stock_adb() { # legacy adb.sh semantics: add adb beside ACM; on failure or lost ACM restore serial; bounded keepalive
   102	  local rc
   103	  sh "$REL/share/mk20-add-adb.sh" >> "$LOGDIR/gadget.log" 2>&1; rc=$?
   104	  [ $rc = 0 ] || log WARN adb "add-adb rc=$rc -> restore serial"
   105	  if [ $rc != 0 ] || [ ! -e "$R/dev/ttyGS0" ] || [ ! -e "$C/acm.usb0" ]; then
   106	    killall adbd 2>/dev/null; [ -x "$R/data/setusbconfig" ] && "$R/data/setusbconfig" serial >> "$LOGDIR/gadget.log" 2>&1; fi
   107	  [ "${MK20_ADB_KEEPALIVE:-1}" = 1 ] || return 0
   108	  ( trap '' HUP INT; exec 5>"$TM/adb-keepalive.lock"; flock -n 5 || exit 0; n=0
   109	    while [ $n -lt 5 ]; do sleep 30
   110	      [ -f "$S/mk20-no-adb" ] && exit 0
   111	      if [ ! -e "$C/ffs.adb" ] && pgrep -f KeyboardDevice >/dev/null; then sleep 10; n=$((n+1)); log INFO adb "keepalive re-add #$n"
   112	        sh "$REL/share/mk20-add-adb.sh" >> "$LOGDIR/gadget.log" 2>&1 || { killall adbd 2>/dev/null; [ -x "$R/data/setusbconfig" ] && "$R/data/setusbconfig" serial; }; fi
   113	    done ) </dev/null >/dev/null 2>&1 6>&- 7>&- 8>&- 9>&- &
   114	}

   116	case "${1:-}" in
   117	boot)
   118	  cfg_load
   119	  log_rotate "$LOGDIR/gadget.log"
   120	  log INFO boot "rel=$REL cfg=$CFG_SRC subnet=$ECM_SUBNET composite=$FEATURE_COMPOSITE hid=$FEATURE_HID adb=$FEATURE_ADB warn=$CFG_WARN"
   121	  i=0; until [ -n "$(cat "$G/UDC" 2>/dev/null)" ] && [ -e "$C/acm.usb0" ] && [ -e "$R/dev/ttyGS0" ] && pgrep -f KeyboardDevice >/dev/null; do
   122	    i=$((i+1)); [ $i -gt 120 ] && { log WARN boot "KD/g1 not up after 120 s"; break; }; sleep 1; done
   123	  sleep "${MK20_SETTLE:-20}"
   124	  MODE=composite; why=""
   125	  NOHID=0; [ "$FEATURE_HID" = 1 ] && kernel_has hid || NOHID=1
   126	  NOADB=0; [ "$FEATURE_ADB" = 1 ] || NOADB=1   # composite only; the stock fallback always adds adb (FAT flag aside)
   127	  [ "$FEATURE_COMPOSITE" = 1 ] || { MODE=stock; why="config/flag"; }
   128	  kernel_has ecm || { MODE=stock; why="${why:+$why,}kernel '$(kernel_line | cut -d' ' -f2)' lacks ecm"; }
   129	  vc=$(vendor_check) || { MODE=stock; why="${why:+$why,}vendor $vc"; }
   130	  [ -n "$(kernel_line)" ] || { MODE=none; why="kernel $(p4sha) not in kernel/ALLOWLIST"; }
   131	  rm -f "$TM/gadget.env"
   132	  if [ $MODE = composite ]; then
   133	    ( umask 077; printf 'MK20_ECM_IP=%s\nMK20_ECM_MASK=%s\nMK20_DEV_MAC=%s\nMK20_HOST_MAC=%s\nMK20_BRIDGE_PORT=%s\nMK20_FLAP_DROPS=%s\nMK20_FLAP_WINDOW=%s\nMK20_NOHID=%s\nMK20_NOADB=%s\n' \
   134	      "$ECM_IP" "$ECM_MASK" "$ECM_DEV_MAC" "$ECM_HOST_MAC" "$BRIDGE_PORT" "$FLAP_DROPS" "$FLAP_WINDOW_S" "$NOHID" "$NOADB" > "$TM/gadget.env.tmp" ) \
   135	      && mv -f "$TM/gadget.env.tmp" "$TM/gadget.env"
   136	    cp -f "$REL/share/gadget.sh" "$TM/gadget-boot.sh"
   137	    MK20_REL=$REL MK20_GADGET_LOG=$LOGDIR/gadget.log MK20_ECM_IP=$ECM_IP MK20_ECM_MASK=$ECM_MASK \
   138	      MK20_DEV_MAC=$ECM_DEV_MAC MK20_HOST_MAC=$ECM_HOST_MAC MK20_BRIDGE_PORT=$BRIDGE_PORT \
   139	      MK20_FLAP_DROPS=$FLAP_DROPS MK20_FLAP_WINDOW=$FLAP_WINDOW_S MK20_NOHID=$NOHID MK20_NOADB=$NOADB \
   140	      sh "$TM/gadget-boot.sh" --boot >> "$LOGDIR/gadget.log" 2>&1
   141	    grc=$?
   142	    log INFO gadget "rc=$grc status=$(cat "$S/gadget-ecm-hid.status" 2>/dev/null) nohid=$NOHID noadb=$NOADB ip=$ECM_IP"
   143	  elif [ $MODE = stock ]; then
   144	    SADB=1; [ -f "$S/mk20-no-adb" ] && SADB=0; kernel_has adb || SADB=0
   145	    log WARN boot "composite skipped: $why -> stock ACM+KD$([ $SADB = 1 ] && echo +adb)"
   146	    [ $SADB = 1 ] && stock_adb
   147	  else log ERROR boot "$why -> no USB changes (stock ACM+KD only)"; fi
   148	  sleep "${MK20_STABLE:-30}"
   149	  # a gadget failsafe (flap rollback + adb re-add, up to ~2.5 min) in progress is not a failure: wait it out (cap)
   150	  # The marker is set by flapmon BEFORE it launches a rollback and by cleanup(); a sample taken while (or just
   151	  # before) it appears is retaken after the failsafe ends. Total wait capped (stays well under the loader deadline).
   152	  w=0; busy_wait() { while [ -f "$TM/gadget.busy" ] && [ $w -lt "${MK20_BUSY_CAP:-150}" ]; do sleep 5; w=$((w+5)); done; }
   153	  busy_wait; try=0; src=1; grc=${grc:-0}
   154	  while [ $try -lt 3 ]; do
   155	    try=$((try+1)); r=$(selftest); src=$?
   156	    if [ -f "$TM/gadget.busy" ] && [ $w -lt "${MK20_BUSY_CAP:-150}" ]; then busy_wait; try=$((try-1)); continue; fi
   157	    [ $src = 0 ] && break; [ $try = 3 ] || sleep "${MK20_SELFTEST_GAP:-10}"
   158	  done
   159	  [ $w -gt 0 ] && log INFO selftest "waited ${w}s for gadget failsafe"
   160	  # On probation, a composite that failed its own setup (gadget rc!=0: ip/alias/bridge/bind -> die -> stock) is a
   161	  # release failure, not "degraded": roll back. Host-caused flap rollbacks happen only after rc 0 and stay WARN.
   162	  if [ $MODE = composite ] && [ "$grc" != 0 ] && [ "$(cat "$MK20_HOME/state/pending" 2>/dev/null)" = "$V" ]; then
   163	    r="$r
   164	FAIL composite setup rc=$grc during probation"; src=1; fi
   165	  echo "$r" > "$TM/selftest.txt"; log INFO selftest "rc=$src try=$try mode=$MODE $(echo "$r" | tr '\n' ';')"
   166	  # per-boot records: selftest-last.json + selftest-1..4.json (previous boots)
   167	  ST=$MK20_HOME/state; for k in 3 2 1; do [ -f "$ST/selftest-$k.json" ] && mv -f "$ST/selftest-$k.json" "$ST/selftest-$((k+1)).json"; done
   168	  [ -f "$ST/selftest-last.json" ] && mv -f "$ST/selftest-last.json" "$ST/selftest-1.json"
   169	  # decide on the 3-sample result ($src) BEFORE the (slower) health record, which can never change it
   170	  if [ $src = 0 ]; then
   171	    if [ -f "$TM/rescue-loader.sh" ]; then sh "$TM/rescue-loader.sh" confirm "$V" && log INFO confirm "ok" || log INFO confirm "not pending/refused"; fi
   172	  else log ERROR selftest "FAILED: not confirming; restoring stock ACM+KD+adb now"
   173	    if [ -f "$TM/gadget-boot.sh" ] && composite_up; then
   174	      # same contract as the confirm waiter: retry once (rc 3 = stuck, already escalated), then persist stock + reboot
   175	      rbn() { MK20_GEN='' MK20_RB_CALLER_RETRIES=1 MK20_REL=$REL MK20_GADGET_LOG=$LOGDIR/gadget.log sh "$TM/gadget-boot.sh" --rollback-now >> "$LOGDIR/gadget.log" 2>&1 6>&- 8>&- 9>&-; }
   176	      rbn; rrc=$?
   177	      if [ $rrc != 0 ] && [ $rrc != 3 ]; then log WARN selftest "rollback rc=$rrc; retrying once"; sleep "${MK20_RB_RETRY_S:-20}"; rbn; rrc=$?; fi
   178	      if [ $rrc != 0 ] && [ $rrc != 3 ]; then log ERROR selftest "rollback failed twice (rc=$rrc) -> mk20-no-composite + reboot"
   179	        touch "${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite}" && touch "${MK20_FLAP_REBOOT:-/mnt/SDCARD/cmd-reboot}"; sync; fi
   180	    fi; fi
   181	  ( health ) > "$ST/selftest-last.json" 2>/dev/null
   182	  if [ "${MK20_MAINT:-1}" = 1 ]; then
   183	    export MK20_REL="$REL" MK20_VERSION="$V"
   184	    ( trap '' HUP INT; exec sh "$REL/sbin/mk20d" maint </dev/null >/dev/null 2>&1 6>&- 7>&- 8>&- 9>&- ) &
   185	  fi
   186	  ;;
   187	health) health;;
   188	maint)
   189	  exec 5>"$TM/mk20d-maint.lock"; flock -n 5 || exit 0   # single instance for the lifetime of the loop
   190	  log INFO maint "start pid=$$"
   191	  while :; do
   192	    cfg_load
   193	    log_rotate "$LOGDIR/mk20d.log"; log_rotate "$LOGDIR/gadget.log"
   194	    b=$TM/bridge.log; [ -f "$b" ] && [ "$(wc -c < "$b")" -gt 262144 ] && { tail -c 65536 "$b" > "$b.t" && cat "$b.t" > "$b"; rm -f "$b.t"; }   # O_APPEND writers keep appending
   195	    health > /dev/null 2>&1
   196	    sleep "${MK20_MAINT_S:-300}" 5>&-   # the lock dies with this loop, not with an orphaned sleep
   197	  done;;
   198	selftest) cfg_load; selftest; exit $?;;
   199	config) cfg_load; cfg_print;;
   200	version) echo "$V";;
   201	*) echo "usage: mk20d boot|health|selftest|config|maint|version"; exit 2;;
   202	esac
