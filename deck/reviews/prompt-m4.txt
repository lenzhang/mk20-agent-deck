+#!/bin/sh
+# macOS host side of the MK20 ECM link (M4). Touches ONLY the network service whose hardware address is the MK20
+# host MAC (default 02:11:22:33:44:56, set by the firmware's ECM host_addr); every other service is left alone.
+#   tools/mk20-host-net.sh status          service, address, route + TUN check, reachability (read-only)
+#   tools/mk20-host-net.sh apply           manual 172.30.250.1/24 on the MK20 service (after the preflight)
+#   tools/mk20-host-net.sh revert          back to the pre-1.1.0 address 10.20.0.1/24
+# Preflight (apply): refuses if another interface already has a route of /24 or narrower inside 172.30.250.0/24
+# (then use the link-local fallback 169.254.7.2 instead). TUN safety: after apply, `route get 172.30.250.2` must
+# resolve to the MK20 interface (mihomo excludes 172.16.0.0/12 from its TUN), else the change is reverted.
+# Env: MK20_HOST_MAC, MK20_SUBNET (a.b.c, default 172.30.250).
+set -eu
+MAC=${MK20_HOST_MAC:-02:11:22:33:44:56}
+NET=${MK20_SUBNET:-172.30.250}
+echo "$NET" | grep -Eq '^[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}$' || { echo "bad MK20_SUBNET" >&2; exit 2; }
+dev=$(networksetup -listallhardwareports | awk -v m="$MAC" '/^Device:/{d=$2} /^Ethernet Address:/{ if (tolower($3)==tolower(m)) print d }' | head -n 1)
+[ -n "$dev" ] || { echo "MK20 interface (MAC $MAC) not present - is the composite up?" >&2; exit 1; }
+svc=$(networksetup -listnetworkserviceorder | awk -v d="$dev" '/^\([0-9*]+\) /{sub(/^\([0-9*]+\) /,""); s=$0} index($0, "Device: " d ")"){print s; exit}')
+[ -n "$svc" ] || { echo "no network service for $dev" >&2; exit 1; }
+route_if() { route -n get "$1" 2>/dev/null | awk '/interface:/{print $2}'; }
+overlap() { # routes /24-or-narrower inside $NET.0/24 on interfaces other than $dev
+  netstat -rn -f inet | awk -v n="$NET" -v d="$dev" '
+    { dst=$1; ifc=$4; if (ifc==d) next
+      split(dst, a, "/"); ip=a[1]; plen=(a[2]=="" ? 32 : a[2])
+      if (index(ip".", n".")==1 && plen>=24) print }'
+}
+show() {
+  echo "service='$svc' device=$dev mac=$MAC"
+  networksetup -getinfo "$svc" | grep -E '^(IP address|Subnet mask):'
+  echo "route $NET.2 -> $(route_if "$NET.2")   route 169.254.7.2 -> $(route_if 169.254.7.2)"
+  ping -c 1 -t 2 "$NET.2" >/dev/null 2>&1 && echo "ping $NET.2 ok" || echo "ping $NET.2 FAILED"
+}
+case "${1:-status}" in
+status) show;;
+apply)
+  o=$(overlap); [ -z "$o" ] || { echo "REFUSED: existing route(s) overlap $NET.0/24 on other interfaces:"; echo "$o"; echo "use link-local 169.254.7.2"; exit 3; }
+  old=$(networksetup -getinfo "$svc" | awk -F': ' '/^IP address/{print $2}')
+  networksetup -setmanual "$svc" "$NET.1" 255.255.255.0
+  i=0; while [ "$(route_if "$NET.2")" != "$dev" ] && [ $i -lt 10 ]; do sleep 1; i=$((i+1)); done
+  if [ "$(route_if "$NET.2")" != "$dev" ]; then
+    echo "route to $NET.2 does not use $dev (TUN/other route?) -> reverting to ${old:-10.20.0.1}"
+    networksetup -setmanual "$svc" "${old:-10.20.0.1}" 255.255.255.0; exit 4; fi
+  show;;
+revert) networksetup -setmanual "$svc" 10.20.0.1 255.255.255.0; show;;
+*) echo "usage: $0 status|apply|revert" >&2; exit 2;;
+esac
