   140	# Transition mutex (1.1.4, r5): ONE flock (fd 7 on /tmp/mk20/transition.lock) is held for the WHOLE setup or the WHOLE
   141	# rollback, so they can never overlap (flapmon, the confirm waiter and mk20d may all ask for --rollback-now). Every
   142	# process spawned while it is held gets 7>&- (adbd, bridge loop, KD, add-adb, flapmon, confirm waiter), so only the
   143	# owner holds it and it is released when the owner exits. A rollback that finds stock verified is a no-op: exit 0
   144	# always means stock is up. Every lock step is checked (fail closed: no mutation without the mutex).
   145	# Owner-local bounded supervisor: the rollback runs cleanup in its own child and waits at most MK20_RB_TIMEOUT
   146	# (default 330 s); on timeout it stops, kills and reaps ITS OWN child tree before releasing the mutex. No caller ever
   147	# kills another owner; a waiter that cannot get the mutex in MK20_RB_LOCK_TRIES*0.25 s (default 900 s) gives up (rc 4).

   150	stock_ok() { [ -e $C/acm.usb0 ] && [ ! -e $C/ecm.usb0 ] && [ ! -e $C/hid.usb0 ] && [ -n "$(udc_now)" ] &&

   153	rb_status() { echo "$1" > /mnt/SDCARD/gadget-ecm-hid.status 2>/dev/null; }
   154	pstart() { awk '$3 != "Z" {print $22}' "/proc/$1/stat" 2>/dev/null; }   # empty if gone or a zombie
   155	# tl_lock <tries>: take the transition mutex on fd 7 (tries 0.25 s apart; BusyBox flock has no -w). 1 = not held.
   156	tl_lock() {
   157	  mkdir -p "$(dirname "$TL")" 2>/dev/null || return 1
   158	  touch "$TL" 2>/dev/null || return 1   # an exec redirect error would exit the shell (special builtin)
   159	  exec 7>"$TL" || return 1
   160	  n=0; until flock -n 7; do n=$((n+1)); [ "$n" -ge "$1" ] && { exec 7>&-; return 1; }; sleep 0.25; done

   162	kids() { for d in /proc/[0-9]*; do [ "$(awk '/^PPid:/{print $2}' "$d/status" 2>/dev/null)" = "$1" ] && echo "${d#/proc/}"; done; }
   163	# kill_tree prints "pid:starttime" for every process it stops (taken while STOPped, before SIGKILL reparents the
   164	# descendants to init) so bounded_wait can check the WHOLE tree afterwards (r6: an unkillable grandchild may hold fd 7).
   165	kill_tree() { kill -STOP "$1" 2>/dev/null; echo "$1:$(pstart "$1")"; for k in $(kids "$1"); do kill_tree "$k"; done; kill -9 "$1" 2>/dev/null; }
   166	# any_alive "<pid:start ...>": 0 if any listed process still runs with the same start time (not a zombie)
   167	any_alive() { for e in $1; do [ -n "${e#*:}" ] && [ "$(pstart "${e%%:*}")" = "${e#*:}" ] && return 0; done; return 1; }
   168	# uint_cap <value> <default> <min> <max>: a validated, clamped decimal (no octal, no overflow: >6 digits = max).
   169	# Every timeout/lock-try knob goes through this BEFORE any lock or fork (r5: bad arithmetic must not kill the owner).
   170	uint_cap() {
   171	  case "$1" in ''|*[!0-9]*) echo "$2"; return;; esac
   172	  v=${1#"${1%%[!0]*}"}; [ -n "$v" ] || v=0
   173	  [ ${#v} -gt 6 ] && { echo "$4"; return; }
   174	  [ "$v" -lt "$3" ] && v=$3; [ "$v" -gt "$4" ] && v=$4; echo "$v"
   175	}
   176	# r5 #5: cleanup worst case on a HEALTHY board = kd_stop<=32 + umount<=10 + kd_start<=23 + readd_adb (g1 wait<=60,
   177	# 5, add-adb ~4 x7, 6x15 re-add window) ~= 260 s -> 330 s so a healthy rollback incl. add-adb is never cut off.
   178	RBRETRY=$(uint_cap "${MK20_RB_RETRY_S:-}" 20 0 300)       # queued/manual rollback: pause before its one retry (s)
   179	RBT=$(uint_cap "${MK20_RB_TIMEOUT:-}" 330 1 900)          # rollback / post-timeout cleanup bound (s)
   180	RBTRIES=$(uint_cap "${MK20_RB_LOCK_TRIES:-}" 4400 1 4800)  # rollback mutex wait (x0.25 s = 1100 s) > worst owner hold:
   181	# setup supervisor SUT+RBT+30 verify+2x10 kill checks+20 escalate (~850 s) + fork/poll overhead (r8 #4)
   182	SUTRIES=$(uint_cap "${MK20_SETUP_LOCK_TRIES:-}" 40 1 4800) # setup mutex wait (x0.25 s)
   183	SUT=$(uint_cap "${MK20_SETUP_TIMEOUT:-}" 450 1 1800)       # whole setup incl. its own failsafe (~60 s + cleanup <=330 s)
   184	# bounded_wait <pid> <s>: 0 = the child exited in time (rc in BW_RC); 1 = timed out: its whole tree was stopped,
   185	# SIGKILLed and the child reaped; 2 = timed out AND still alive 10 s after SIGKILL (D state, e.g. a stuck UDC write):
   186	# not waited for (an unbounded wait would wedge us). "Alive" covers EVERY process of the tree snapshotted by
   187	# kill_tree, not just the root (r6). Only ever used on OUR OWN children.
   188	bounded_wait() {
   189	  bst=$(pstart "$1"); bi=0; BW_RC=1
   190	  while [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ] && [ $bi -lt $(($2*4)) ]; do sleep 0.25; bi=$((bi+1)); done
   191	  if [ -n "$bst" ] && [ "$(pstart "$1")" = "$bst" ]; then
   192	    bsnap="$1:$bst $(kill_tree "$1")"; bi=0
   193	    while any_alive "$bsnap" && [ $bi -lt 40 ]; do sleep 0.25; bi=$((bi+1)); done
   194	    any_alive "$bsnap" && return 2
   195	    wait "$1" 2>/dev/null; return 1
   196	  fi
   197	  wait "$1" 2>/dev/null; BW_RC=$?; return 0
   198	}
   199	# escalate <why>: no transition can complete (an unkillable child holds fd 7, or stock is not reached): persist
   200	# stock mode (flag read at boot) and hard-reset via the rescue watcher (cmd-reboot) - same path as flapmon's.
   201	# r6: the writes run in a bounded child WITHOUT fd 7 (a hung FAT write cannot wedge the mutex); success = both files
   202	# exist afterwards (sync cannot mask a failed touch). 0 = escalated, 1 = could not persist (caller exits 1, not 3).
   203	# r8: BW_RC is saved/restored (callers still need their own child's rc); the writer marks success on tmpfs BEFORE
   204	# sync (a slow sync or the watcher consuming cmd-reboot is not a failure); the STUCK/ESCALATED status is written last.
   205	escalate() {
   206	  ebrc=${BW_RC:-1}; EF=${MK20_FLAP_FLAG:-/mnt/SDCARD/mk20-no-composite}; ER=${MK20_FLAP_REBOOT:-/mnt/SDCARD/cmd-reboot}
   207	  EOKF=/tmp/mk20/escalate.ok; rm -f "$EOKF"

   209	  ( touch "$EF" && touch "$ER" && [ -e "$EF" ] && [ -e "$ER" ] && touch "$EOKF"; sync ) </dev/null 7>&- &
   210	  bounded_wait $! 20; ebw=$?; BW_RC=$ebrc
   211	  if [ -e "$EOKF" ]; then rm -f "$EOKF"; rb_status "$2"; return 0; fi
